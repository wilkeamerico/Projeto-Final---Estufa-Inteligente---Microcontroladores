function desenharOLED(d){
  const cv = document.getElementById('oledCanvas');
  const ctx = cv.getContext('2d');
  ctx.fillStyle = '#04090f';
  ctx.fillRect(0, 0, cv.width, cv.height);

  ctx.fillStyle = 'rgba(255,255,255,0.025)';
  for(let y=0; y<128; y+=2){ ctx.fillRect(0, y, 256, 1); }

  ctx.fillStyle = '#dff6ff';
  ctx.font = 'bold 14px ui-monospace, SFMono-Regular, Menlo, Consolas, monospace';
  ctx.textBaseline = 'top';

  if(d.dhtOk){
    ctx.fillText(`T:${d.temp.toFixed(1)}C (L:${Math.round(d.tempLimite)})`, 2, 2);
    ctx.fillText(`U:${d.umid.toFixed(1)}% (L:${Math.round(d.umidadeMin)})`, 2, 18);
  } else {
    ctx.fillText('T: ERRO DHT22!', 2, 2);
    ctx.fillText('U: ERRO DHT22!', 2, 18);
  }

  ctx.fillText(`Gas:${d.gas}`, 2, 42);
  ctx.fillText(`Lz:${d.lampLig ? 'ON' : 'OFF'}`, 134, 42);

  ctx.fillText(`Col:${d.coolerLig ? '[ON]' : '[OFF]'}`, 2, 70);
  ctx.fillText(`Bmb:${d.bombaLig ? '[ON]' : '[OFF]'}`, 134, 70);

  let statusMsg = 'SISTEMA OK';
  if(!d.dhtOk) statusMsg = 'ERRO SENSOR DHT22';
  else if(d.critico) statusMsg = 'ALARM: CRITICO!!';
  else if(d.buzzerStatus === 1) statusMsg = 'ALARM: TEMP ALTA';
  else if(d.buzzerStatus === 2) statusMsg = 'ALARM: UMID BAIXA';

  ctx.fillStyle = (d.critico || !d.dhtOk) ? '#fbbf24' : '#dff6ff';
  ctx.fillText(statusMsg, 2, 102);
}

// ---------------- PROCESSA NOVOS CICLOS SINCRONIZADOS ----------------
function registrarCicloSincronizado(item, tempLimite){
  if(item.ciclo <= ultimoCicloWeb) return;
  ultimoCicloWeb = item.ciclo;

  // 1) Console Serial Web (idêntico ao Arduino IDE)
  if(serialHistorico.length === 0 && document.getElementById('serialConsole').textContent.startsWith('Aguardando')){
    document.getElementById('serialConsole').textContent = '';
  }
  serialHistorico.push(item.serialLog);
  if(serialHistorico.length > 40) serialHistorico.shift();
  const sc = document.getElementById('serialConsole');
  sc.textContent = serialHistorico.join('\n');
  if(autoScroll) sc.scrollTop = sc.scrollHeight;

  // 2) Tabela de Registros
  const hora = new Date().toLocaleTimeString('pt-BR');
  const bombaEmerg  = item.critico;
  const coolerEmerg = item.critico || (item.dhtOk && item.temp > tempLimite);

  let alarmeTxt = 'SISTEMA OK';
  if(!item.dhtOk) alarmeTxt = 'ERRO DHT22';
  else if(item.critico) alarmeTxt = 'CRÍTICO';
  else if(item.buzzerStatus === 1) alarmeTxt = 'TEMP ALTA';
  else if(item.buzzerStatus === 2) alarmeTxt = 'UMID BAIXA';

  registrosTabela.unshift({
    ciclo: item.ciclo,
    hora: hora,
    temp: item.dhtOk ? item.temp.toFixed(1) : 'ERRO',
    umid: item.dhtOk ? item.umid.toFixed(1) : 'ERRO',
    lum: item.lum,
    gas: item.gas,
    luz: item.lampLig ? 'LIGADA' : 'DESLIGADA',
    cooler: coolerEmerg ? 'EMERGÊNCIA' : (item.coolerLig ? 'LIGADO' : 'DESLIGADO'),
    bomba: bombaEmerg ? 'EMERGÊNCIA' : (item.bombaLig ? 'LIGADA' : 'DESLIGADA'),
    alarme: alarmeTxt
  });
  if(registrosTabela.length > 40) registrosTabela.pop();

  const tb = document.getElementById('tabelaRegistros');
  const tr = document.createElement('tr');
  tr.innerHTML =
    `<td>#${item.ciclo}</td>` +
    `<td>${hora}</td>` +
    `<td>${item.dhtOk ? item.temp.toFixed(1) + ' °C' : 'ERRO'}</td>` +
    `<td>${item.dhtOk ? item.umid.toFixed(1) + ' %' : 'ERRO'}</td>` +
    `<td>${item.lum}</td>` +
    `<td>${item.gas}</td>` +
    `<td>${pillAtuadorTabela(item.lampLig, false, true)}</td>` +
    `<td>${pillAtuadorTabela(item.coolerLig, coolerEmerg, false)}</td>` +
    `<td>${pillAtuadorTabela(item.bombaLig, bombaEmerg, true)}</td>` +
    `<td>${alarmeTxt}</td>`;
  tb.insertBefore(tr, tb.firstChild);
  while(tb.children.length > 40) tb.removeChild(tb.lastChild);
}

  atualizarAtuadorCard('cardActBomba', 'badgeBomba', 'modeBomba', d.bombaLig, d.bombaManual, bombaEmerg, true);

  ['btnBombaOn','btnBombaOff','btnBombaAuto'].forEach(id=>document.getElementById(id).disabled = bombaEmerg);
  const lbEl = document.getElementById('lockBomba');
  if(bombaEmerg){ lbEl.style.display='block'; lbEl.textContent='EMERGÊNCIA: Alarme Crítico mantendo Bomba acionada.'; }
  else { lbEl.style.display='none'; }

  ['btnCoolerOn','btnCoolerOff','btnCoolerAuto'].forEach(id=>document.getElementById(id).disabled = coolerEmerg);
  const lcEl = document.getElementById('lockCooler');

  else { lcEl.style.display='none'; }

  // Gêmeo Digital OLED
  desenharOLED(d);

  // Processa todos os novos ciclos enviados pelo buffer do ESP32 (sem perder nenhum!)

    d.novos.forEach(item => registrarCicloSincronizado(item, d.tempLimite));
  }

// =========================================================================
//  FUNÇÃO AUXILIAR PARA JSON
// =========================================================================
String jsonEscape(const String& s) {
  String out;
  out.reserve(s.length() + 16);
  for (size_t i = 0; i < s.length(); i++) {
    char c = s[i];
    if (c == '\\') out += "\\\\";
    else if (c == '\"') out += "\\\"";
    else if (c == '\n') out += "\\n";
    else if (c == '\r') out += "\\r";
    else if (c == '\t') out += "\\t";
    else out += c;
  }
  return out;
}

// =========================================================================
//  LÓGICA DE ALARMES E ATUADORES (EXTRAÍDA PARA GARANTIR SINCRONIA)
// =========================================================================
void atualizarAtuadoresEAlarmes(unsigned long atualMillis) {
  if (dhtValido && temperatura > tempCritica && umidade < umidCritica) {
    alarmeCriticoTrava = true;
  }

  if (alarmeCriticoTrava) {
    buzzerStatus = 3;
    tone(buzzer, 1000);
    digitalWrite(RELE_COOLER, LOW);  // FORÇA LIGAR
    digitalWrite(RELE_BOMBA, LOW);   // FORÇA LIGAR
  } else if (dhtValido && temperatura > tempLimite) {
    buzzerStatus = 1;
    digitalWrite(RELE_COOLER, LOW);  // Força Cooler LIGADO

    if (atualMillis - ultimoBip5s >= 5000) {
      ultimoBip5s = atualMillis;
      tone(buzzer, 1200, 200);
    }

    if (bombaManual) digitalWrite(RELE_BOMBA, bombaModoWeb ? LOW : HIGH);
    else digitalWrite(RELE_BOMBA, (umidade < umidadeMin) ? LOW : HIGH);
  } else if (dhtValido && umidade < umidadeMin) {
    buzzerStatus = 2;

    if (estagioBipDuplo == 0 && atualMillis - ultimoBip10s >= 10000) {
      ultimoBip10s = atualMillis;
      tone(buzzer, 1000, 150);
      estagioBipDuplo = 1;
    } else if (estagioBipDuplo == 1 && atualMillis - ultimoBip10s >= 250) {
      tone(buzzer, 1000, 150);
      estagioBipDuplo = 0;
    }

    if (coolerManual) digitalWrite(RELE_COOLER, coolerModoWeb ? LOW : HIGH);
    else digitalWrite(RELE_COOLER, HIGH);

    if (bombaManual) digitalWrite(RELE_BOMBA, bombaModoWeb ? LOW : HIGH);
    else digitalWrite(RELE_BOMBA, LOW);
  } else {
    if (buzzerStatus != 0) {
      noTone(buzzer);
    }
    buzzerStatus = 0;
    estagioBipDuplo = 0;

    if (coolerManual) digitalWrite(RELE_COOLER, coolerModoWeb ? LOW : HIGH);
    else digitalWrite(RELE_COOLER, HIGH);

    if (bombaManual) digitalWrite(RELE_BOMBA, bombaModoWeb ? LOW : HIGH);
    else digitalWrite(RELE_BOMBA, HIGH);
  }

  // Controle da Iluminação
  if (lampadaManual) {
    digitalWrite(RELE_LAMPADA, lampadaModoWeb ? LOW : HIGH);
  } else {
    digitalWrite(RELE_LAMPADA, (luminosidade < luminMin) ? LOW : HIGH);
  }
}

void adicionarRegistroHistorico(const RegistroCiclo& reg) {
  historico[histHead] = reg;
  histHead = (histHead + 1) % MAX_HIST;
  if (histCount < MAX_HIST) histCount++;
}

void handleApiStatus() {
  unsigned long desde = server.hasArg("desde") ? (unsigned long)server.arg("desde").toInt() : 0;

  bool bombaLig  = (digitalRead(RELE_BOMBA) == LOW);
  bool coolerLig = (digitalRead(RELE_COOLER) == LOW);
  bool lampLig   = (digitalRead(RELE_LAMPADA) == LOW);

  String json = "{";
  json += "\"temp\":" + String(temperatura, 1) + ",";
  json += "\"umid\":" + String(umidade, 1) + ",";
  json += "\"dhtOk\":" + String(dhtValido ? "true" : "false") + ",";
  json += "\"lum\":" + String(luminosidade) + ",";
  json += "\"gas\":" + String(nivelGas) + ",";
  json += "\"tempLimite\":" + String(tempLimite, 1) + ",";
  json += "\"umidadeMin\":" + String(umidadeMin, 0) + ",";
  json += "\"tempCritica\":" + String(tempCritica, 1) + ",";
  json += "\"umidCritica\":" + String(umidCritica, 0) + ",";
  json += "\"luminMin\":" + String(luminMin) + ",";
  json += "\"gasLimite\":" + String(gasLimite) + ",";
  json += "\"buzzerStatus\":" + String(buzzerStatus) + ",";
  json += "\"critico\":" + String(alarmeCriticoTrava ? "true" : "false") + ",";
  json += "\"lampLig\":" + String(lampLig ? "true" : "false") + ",";
  json += "\"lampManual\":" + String(lampadaManual ? "true" : "false") + ",";
  json += "\"bombaLig\":" + String(bombaLig ? "true" : "false") + ",";
  json += "\"bombaManual\":" + String(bombaManual ? "true" : "false") + ",";
  json += "\"coolerLig\":" + String(coolerLig ? "true" : "false") + ",";
  json += "\"coolerManual\":" + String(coolerManual ? "true" : "false") + ",";
  json += "\"ciclo\":" + String(cicloLeitura) + ",";

  // Envia todos os registros em ordem cronológica que a Web ainda não recebeu
  json += "\"novos\":[";
  bool primeiro = true;
  int startIdx = (histCount < MAX_HIST) ? 0 : histHead;
  for (int i = 0; i < histCount; i++) {
    int idx = (startIdx + i) % MAX_HIST;
    const RegistroCiclo& r = historico[idx];
    if (r.ciclo > desde) {
      if (!primeiro) json += ",";
      primeiro = false;
      json += "{";
      json += "\"ciclo\":" + String(r.ciclo) + ",";
      json += "\"temp\":" + String(r.temp, 1) + ",";
      json += "\"umid\":" + String(r.umid, 1) + ",";
      json += "\"dhtOk\":" + String(r.dhtOk ? "true" : "false") + ",";
      json += "\"lum\":" + String(r.lum) + ",";
      json += "\"gas\":" + String(r.gas) + ",";
      json += "\"lampLig\":" + String(r.lampLig ? "true" : "false") + ",";
      json += "\"coolerLig\":" + String(r.coolerLig ? "true" : "false") + ",";
      json += "\"bombaLig\":" + String(r.bombaLig ? "true" : "false") + ",";
      json += "\"critico\":" + String(r.critico ? "true" : "false") + ",";
      json += "\"buzzerStatus\":" + String(r.buzzerStatus) + ",";
      json += "\"serialLog\":\"" + jsonEscape(r.logText) + "\"";
      json += "}";
    }
  }
  json += "]}";

  server.send(200, "application/json", json);
}

void handleSetLimites() {
  if (server.hasArg("tmax"))  tempLimite  = constrain(server.arg("tmax").toFloat(), 0.0f, 80.0f);
  if (server.hasArg("umin"))  umidadeMin  = constrain(server.arg("umin").toFloat(), 0.0f, 100.0f);
  if (server.hasArg("tcrit")) tempCritica = constrain(server.arg("tcrit").toFloat(), 0.0f, 80.0f);
  if (server.hasArg("ucrit")) umidCritica = constrain(server.arg("ucrit").toFloat(), 0.0f, 100.0f);
  if (server.hasArg("lmin"))  luminMin    = constrain(server.arg("lmin").toInt(), 0, 4095);
  if (server.hasArg("gmax"))  gasLimite   = constrain(server.arg("gmax").toInt(), 0, 4095);

  salvarLimitesFlash();
  atualizarAtuadoresEAlarmes(millis());

  if (server.hasArg("ajax")) {
    server.send(200, "application/json", "{\"ok\":true}");
  } else {
    server.sendHeader("Location", "/");
    server.send(303);
  }
}

void handleBomba() {
  if (!alarmeCriticoTrava) {
    String e = server.arg("estado");
    if (e == "on") {
      bombaManual = true;
      bombaModoWeb = true;
    } else if (e == "off") {
      bombaManual = true;
      bombaModoWeb = false;
    } else {
      bombaManual = false;
    }
    atualizarAtuadoresEAlarmes(millis());
  }
  if (server.hasArg("ajax")) {
    server.send(200, "application/json", "{\"ok\":true}");
  } else {
    server.sendHeader("Location", "/");
    server.send(303);
  }
}

void handleLampada() {
  String e = server.arg("estado");

void handleResetAlarme() {
  bool resetOk = false;
  if (!dhtValido || temperatura <= tempCritica || umidade >= umidCritica) {
    alarmeCriticoTrava = false;
    noTone(buzzer);
    resetOk = true;
    atualizarAtuadoresEAlarmes(millis());
  }
  if (server.hasArg("ajax")) {
    server.send(200, "application/json", String("{\"resetOk\":") + (resetOk ? "true" : "false") + "}");
  } else {
    server.sendHeader("Location", "/");
    server.send(303);
  }
}

void setup() {
  Serial.begin(115200);
  delay(500);

  carregarLimites();

  Wire.begin();
  if (!display.begin(SSD1306_SWITCHCAPVCC, OLED_ADDRESS)) {
    Serial.println("Falha ao iniciar OLED SSD1306!");
    while (1);
  }
  display.clearDisplay();
  display.setTextColor(SSD1306_WHITE);
  display.setTextSize(1);
  display.setCursor(0, 0);
  display.println("Estufa Inteligente");
  display.println("Sensor: DHT22");
  display.println("Iniciando AP...");
  display.display();

  dht.begin();

  pinMode(RELE_BOMBA, OUTPUT);
  digitalWrite(RELE_BOMBA, HIGH);
  pinMode(RELE_COOLER, OUTPUT);
  digitalWrite(RELE_COOLER, HIGH);
  pinMode(RELE_LAMPADA, OUTPUT);
  digitalWrite(RELE_LAMPADA, HIGH);
  pinMode(buzzer, OUTPUT);

  WiFi.mode(WIFI_AP);
  WiFi.softAPConfig(ap_ip, ap_ip, IPAddress(255, 255, 255, 0));
  WiFi.softAP(ap_ssid, ap_password);

  server.on("/", handleRoot);
  server.on("/api/status", handleApiStatus);
  server.on("/setlimites", handleSetLimites);
  server.on("/bomba", handleBomba);
  server.on("/cooler", handleCooler);
  server.on("/lampada", handleLampada);
  server.on("/resetAlarme", handleResetAlarme);
  server.begin();

  ultimoCicloLeitura = millis() - INTERVALO_LEITURA;
}

// =========================================================================
//  LOOP
// =========================================================================
void loop() {
  server.handleClient();
  unsigned long atualMillis = millis();

  // --- PASSO 1: LEITURA REAL DOS SENSORES (A CADA 2 SEGUNDOS) ---
  if (atualMillis - ultimoCicloLeitura >= INTERVALO_LEITURA) {
    ultimoCicloLeitura = atualMillis;
    cicloLeitura++;

    float t = dht.readTemperature();
    float h = dht.readHumidity();

    if (isnan(t) || isnan(h)) {
      dhtValido = false;
    } else {
      dhtValido = true;
      temperatura = t;
      umidade = h;
    }

    luminosidade = analogRead(LDR_PIN);
    nivelGas = analogRead(MQ_PIN);

    // --- PASSO 2: ATUALIZA ALARMES E RELÉS COM OS VALORES RECÉM-LIDOS ---
    atualizarAtuadoresEAlarmes(atualMillis);

    // Lê os estados finais já sincronizados dos atuadores
    bool bombaLig  = (digitalRead(RELE_BOMBA) == LOW);
    bool coolerLig = (digitalRead(RELE_COOLER) == LOW);
    bool lampLig   = (digitalRead(RELE_LAMPADA) == LOW);

    // --- PASSO 3: ATUALIZAÇÃO OLED SSD1306 ---
    display.clearDisplay();
    display.setTextSize(1);
    display.setCursor(0, 0);
    if (dhtValido) {
      display.print("T:");
      display.print(temperatura, 1);
      display.print("C (L:");
      display.print(tempLimite, 0);
      display.println(")");
      display.print("U:");
      display.print(umidade, 1);
      display.print("% (L:");
      display.print(umidadeMin, 0);
      display.println(")");
    } else {
      display.println("T: ERRO DHT22!");
      display.println("U: ERRO DHT22!");
    }

    display.setCursor(0, 20);
    display.print("Gas:");
    display.print(nivelGas);
    display.setCursor(66, 20);
    display.print("Lz:");
    display.print(lampLig ? "ON" : "OFF");

    display.setCursor(0, 34);
    display.print("Col:");
    display.print(coolerLig ? "[ON]" : "[OFF]");
    display.setCursor(66, 34);
    display.print("Bmb:");
    display.print(bombaLig ? "[ON]" : "[OFF]");

    display.setCursor(0, 50);
    if (!dhtValido) display.print("ERRO SENSOR DHT22");
    else if (alarmeCriticoTrava) display.print("ALARM: CRITICO!!");
    else if (buzzerStatus == 1) display.print("ALARM: TEMP ALTA");
    else if (buzzerStatus == 2) display.print("ALARM: UMID BAIXA");
    else display.print("SISTEMA OK");

    display.display();

    // --- PASSO 4: MONTA REGISTRO ÚNICO (SERIAL IDE + BUFFER WEB 1:1) ---
    char buf[512];
    String statusLinha;
    if (!dhtValido) statusLinha = "-> ERRO: Falha na leitura do sensor DHT22!";
    else if (alarmeCriticoTrava) statusLinha = "-> ALARME CRITICO ATIVADO! (TRAVA DE EMERGENCIA)";
    else if (buzzerStatus == 1) statusLinha = "-> ALERTA: Temp Alta (Bip 5s / Cooler em Emergencia)";
    else if (buzzerStatus == 2) statusLinha = "-> ALERTA: Umid Baixa (Bip 10s / Bomba Auto)";
    else statusLinha = "-> Buzzer: Silencioso (Sistema Estavel)";

    if (dhtValido) {
      snprintf(buf, sizeof(buf),
        "=== MONITORAMENTO ESTUFA (CICLO #%lu) ===\n"
        "Temperatura: %.1f C  | Lim: %.1f C | Crit: %.1f C\n"
        "Umidade:     %.1f %% | Min: %.0f %% | Crit: %.0f %%\n"
        "Luminosidade:%d ADC (Min: %d) | Luz: %s\n"
        "CO2/Gas ADC: %d (Lim: %d)\n"
        "Cooler: %s | Bomba: %s\n"
        "%s\n"
        "=========================================\n",
        cicloLeitura,
        temperatura, tempLimite, tempCritica,
        umidade, umidadeMin, umidCritica,
        luminosidade, luminMin, lampLig ? "LIGADA" : "DESLIGADA",
        nivelGas, gasLimite,
        coolerLig ? "LIGADO" : "DESLIGADO",
        bombaLig ? "LIGADA" : "DESLIGADA",
        statusLinha.c_str()
      );
    } else {
      snprintf(buf, sizeof(buf),
        "=== MONITORAMENTO ESTUFA (CICLO #%lu) ===\n"
        "Sensor DHT22: FALHA NA LEITURA (verifique pino 5)\n"
        "Luminosidade:%d ADC (Min: %d) | Luz: %s\n"
        "CO2/Gas ADC: %d (Lim: %d)\n"
        "Cooler: %s | Bomba: %s\n"
        "%s\n"
        "=========================================\n",
        cicloLeitura,
        luminosidade, luminMin, lampLig ? "LIGADA" : "DESLIGADA",
        nivelGas, gasLimite,
        coolerLig ? "LIGADO" : "DESLIGADO",
        bombaLig ? "LIGADA" : "DESLIGADA",
        statusLinha.c_str()
      );
    }

    RegistroCiclo reg;
    reg.ciclo        = cicloLeitura;
    reg.temp         = temperatura;
    reg.umid         = umidade;
    reg.dhtOk        = dhtValido;
    reg.lum          = luminosidade;
    reg.gas          = nivelGas;
    reg.lampLig      = lampLig;
    reg.coolerLig    = coolerLig;
    reg.bombaLig     = bombaLig;
    reg.critico      = alarmeCriticoTrava;
    reg.buzzerStatus = buzzerStatus;
    reg.logText      = String(buf);

    adicionarRegistroHistorico(reg);
    Serial.print(reg.logText);
  } else {
    // Mantém os bipes não-bloqueantes e atuadores atualizados entre ciclos
    atualizarAtuadoresEAlarmes(atualMillis);
  }
}
